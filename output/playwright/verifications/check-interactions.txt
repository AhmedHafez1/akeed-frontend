async (page) => {
  const results = [];
  const assert = (condition, name) => { if (!condition) throw new Error(name); results.push(name); };
  const visit = async (query = '') => {
    await page.goto('http://127.0.0.1:3098/en/verifications' + query);
    await page.getByRole('button', { name: 'All', exact: true }).waitFor();
    await page.locator('[aria-busy="true"]').first().waitFor({state:'detached'});
  };
  const rows = () => page.locator('tbody tr');
  const audit = async () => {
    await page.getByRole('button', {name:'Inspect fixture requests'}).click();
    return await page.getByLabel('Fixture requests').innerText();
  };
  await page.setViewportSize({width:1280,height:1000});
  await visit('?preserved=yes');
  assert(await rows().count() === 6, 'First cursor page contains six rows');
  assert((await audit()).split('\n').every(line => !line.startsWith('POST')), 'Initial render submits no actions');
  const confirmed = rows().filter({hasText:'#1047'});
  assert(await confirmed.getByTitle('Delivered: Completed by customer confirmation', {exact:true}).count() === 1, 'Confirmation completes missing delivery evidence without a fabricated timestamp');
  assert(await confirmed.locator('ol > li > span.bg-emerald-600').count() === 3, 'Confirmed shows three filled green circles');
  assert(await rows().filter({hasText:'#1044'}).locator('ol > li > span.bg-emerald-600').count() === 3, 'Canceled shows three filled green circles');
  assert(await confirmed.getByRole('button').count() === 0, 'Settled row has no actions');
  await page.getByRole('button', {name:'Load more',exact:true}).click();
  await page.getByText('11 verifications loaded',{exact:true}).waitFor();
  assert(await rows().count() === 11, 'Load more appends all remaining rows');
  assert(await rows().filter({hasText:'#1039'}).getByRole('button').count() === 0, 'No-reply without capabilities cannot cancel');
  assert(await rows().filter({hasText:'#1038'}).getByRole('button').count() === 0, 'Failed without capabilities cannot retry');
  await page.getByRole('button', {name:'Failed',exact:true}).click();
  await page.getByText('2 verifications loaded',{exact:true}).waitFor();
  assert(page.url().includes('status=failed') && page.url().includes('preserved=yes'), 'Filter is URL-backed and preserves other query parameters');
  await page.reload();
  await page.getByText('2 verifications loaded',{exact:true}).waitFor();
  assert(await page.getByRole('button',{name:'Failed',exact:true}).getAttribute('aria-pressed') === 'true', 'Refresh restores status');
  await page.getByRole('button',{name:'Confirmed',exact:true}).click();
  await page.getByText('1 verification loaded',{exact:true}).waitFor();
  await page.goBack();
  await page.getByText('2 verifications loaded',{exact:true}).waitFor();
  assert(await page.getByRole('button',{name:'Failed',exact:true}).getAttribute('aria-pressed') === 'true', 'Browser back restores status');
  await page.goForward();
  await page.getByText('1 verification loaded',{exact:true}).waitFor();
  assert(await page.getByRole('button',{name:'Confirmed',exact:true}).getAttribute('aria-pressed') === 'true', 'Browser forward restores status');
  await page.getByRole('button',{name:'All verifications',exact:true}).click();
  await page.getByText('6 verifications loaded',{exact:true}).waitFor();
  assert(!page.url().includes('status='), 'All summary clears status');
  await page.getByRole('button',{name:'Awaiting reply',exact:true}).click();
  await page.getByText('5 verifications loaded',{exact:true}).waitFor();
  assert((await audit()).includes('status=sent%2Cdelivered%2Cread%2Cno_reply'), 'Awaiting reply uses unchanged API grouping');
  await page.getByRole('combobox',{name:'Date range'}).selectOption('today');
  await page.locator('[aria-busy="true"]').first().waitFor({state:'detached'});
  assert((await audit()).includes('date_range=today'), 'Date range reaches API');
  await visit();
  await page.getByRole('button',{name:'Cancel order',exact:true}).click();
  await page.getByRole('button',{name:'Keep order',exact:true}).waitFor();
  assert(!(await audit()).includes('POST '), 'Request cancellation only opens inline confirmation');
  await page.getByRole('button',{name:'Keep order',exact:true}).click();
  assert(!(await audit()).includes('POST '), 'Keep order submits nothing');
  await page.getByRole('button',{name:'Cancel order',exact:true}).click();
  await page.getByRole('button',{name:'Yes, cancel',exact:true}).click();
  await page.getByRole('button',{name:'Cancel order',exact:true}).waitFor({state:'detached'});
  assert((await audit()).includes('POST /api/verifications/triage-0/cancel'), 'Confirm invokes existing cancellation endpoint');
  await page.getByRole('button',{name:'Retry verification',exact:true}).click();
  await page.getByRole('button',{name:'Retry verification',exact:true}).waitFor({state:'detached'});
  assert((await audit()).includes('POST /api/orders/triage-order-3/verification/retry'), 'Retry invokes existing order handler');
  await visit('?scenario=action-error');
  await page.getByRole('button',{name:'Retry verification',exact:true}).click();
  await rows().filter({hasText:'#1045'}).getByRole('alert').waitFor();
  assert(await rows().count() === 6, 'Retry error stays beneath actions and retains records');
  await page.getByRole('button',{name:'Cancel order',exact:true}).click();
  await page.getByRole('button',{name:'Yes, cancel',exact:true}).click();
  await rows().filter({hasText:'#1048'}).getByRole('alert').waitFor();
  assert(await page.getByRole('button',{name:'Keep order',exact:true}).isVisible(), 'Cancellation error preserves inline recovery');
  await visit('?scenario=cursor-error');
  await page.getByRole('button',{name:'Load more',exact:true}).click();
  await page.getByText('Synthetic cursor failure',{exact:true}).waitFor();
  assert(await rows().count() === 6 && await page.getByRole('button',{name:'Load more',exact:true}).isEnabled(), 'Cursor failure retains rows and allows retry');
  await visit('?scenario=readonly');
  assert(await page.getByText(/You have read-only access/).isVisible() && await rows().getByRole('button').count() === 0, 'Read-only notice and no row actions');
  assert(await page.getByRole('button',{name:'Verify order',exact:true}).isDisabled(), 'Read-only prevents manual entry');
  await visit('?scenario=disconnected');
  assert(await page.getByRole('button',{name:'Verify order',exact:true}).isDisabled(), 'Disconnected source prevents manual entry');
  await visit('?scenario=empty');
  assert(await page.getByRole('heading',{name:'No verifications yet',exact:true}).isVisible(), 'First-use empty state');
  assert(await page.locator('input[type="tel"]').count() > 0, 'First-use test verification panel retained');
  await page.getByRole('button',{name:'Failed',exact:true}).click();
  await page.getByText('No verifications match this status and date range.',{exact:true}).waitFor();
  assert(await page.locator('input[type="tel"]').count() === 0, 'Filtered empty state remains distinct');
  await visit('?scenario=stats-error');
  assert(await rows().count() === 6, 'Stats failure retains loaded records');
  assert(!(await page.getByRole('region',{name:'Verification workload'}).innerText()).includes('NaN'), 'Unavailable metrics are not fabricated');
  return results;
}
